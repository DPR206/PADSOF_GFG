private void nombre(){
    Scanner scanner = new Scanner(System.in);
    boolean exitLoop = false;
    while (!appExited && !exitLoop) {
        System.out.println("What do you wish to do? (enter the nº)");
        int i = 1;
        System.out.println("\t[" + i++ + "] ");
        basicLoopPrinter(i);
        chosenOption = scanner.nextInt();

        switch (chosenOption) {
            case 1:
                break;
            case 2:
                break;
            case 2: /* Exit */
                exit();
                break;
            default: /* Go back */
                exitLoop = true;
                break;
        }
    }
}
